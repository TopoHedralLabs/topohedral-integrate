64//{{{ enum: GaussFamily
65/// A supported Gauss quadrature family.
66///
67/// See the [Gaussian-rules guide](https://topohedrallabs.github.io/topohedral-integrate/latest/user-guide/gaussian-rules/).
68#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
69#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
70pub enum GaussFamily {
71    /// Gauss-Legendre quadrature on `[-1, 1]` with unit weight.
72    Legendre,
73    /// Gauss-Lobatto quadrature family on `[-1, 1]`.
74    ///
75    /// Rules retrieved from a [`GaussRuleSet`] include both endpoints.
76    Lobatto,
77}
78
79impl GaussFamily {
80    /// Returns the reference-interval integral of the weight used by this family's node-generating
81    /// recurrence.
82    ///
83    /// For Lobatto rules, this is the integral of the Jacobi `(1, 1)` weight used to generate the
84    /// interior nodes, not the sum of the final Lobatto weights.
85    pub fn weight_integral(&self) -> f64 {
86        match self {
87            Self::Legendre => 2.0,
88            Self::Lobatto => 4.0 / 3.0,
89        }
90    }
91
92    /// Returns this family's reference interval.
93    pub fn range(&self) -> (f64, f64) {
94        match self {
95            Self::Legendre => (-1.0, 1.0),
96            Self::Lobatto => (-1.0, 1.0),
97        }
98    }
99
100    /// Returns the number of points needed for a rule with at least `degree` polynomial exactness.
101    ///
102    /// An `n`-point Legendre rule has exactness `2 * n - 1`, while an `n`-point Lobatto rule has
103    /// exactness `2 * n - 3`. Consequently, an even requested degree selects a rule whose actual
104    /// exactness is one degree higher.
105    fn point_count_for_degree(
106        self,
107        degree: usize,
108    ) -> usize {
109        match self {
110            Self::Legendre => degree / 2 + 1,
111            Self::Lobatto => degree / 2 + 2,
112        }
113    }
114
115    fn exactness_for_point_count(
116        self,
117        point_count: usize,
118    ) -> Option<usize> {
119        match self {
120            Self::Legendre if point_count >= 1 => point_count.checked_mul(2)?.checked_sub(1),
121            Self::Lobatto if point_count >= 2 => point_count.checked_mul(2)?.checked_sub(3),
122            _ => None,
123        }
124    }
125
126    /// Returns the minimum supported point count for this quadrature family.
127    fn minimum_point_count(self) -> usize {
128        match self {
129            Self::Legendre => 1,
130            Self::Lobatto => 2,
131        }
132    }
133
134    fn maximum_point_count(self) -> usize {
135        self.point_count_for_degree(MAX_DEGREE)
136    }
137}
138
139impl fmt::Display for GaussFamily {
140    fn fmt(
141        &self,
142        formatter: &mut fmt::Formatter<'_>,
143    ) -> fmt::Result {
144        match self {
145            Self::Legendre => formatter.write_str("legendre"),
146            Self::Lobatto => formatter.write_str("lobatto"),
147        }
148    }
149}

152/// An error encountered while constructing or retrieving a Gaussian rule.
153///
154/// See the [Gaussian-rules guide](https://topohedrallabs.github.io/topohedral-integrate/latest/user-guide/gaussian-rules/).
155#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
156#[derive(Clone, Debug, Eq, Error, Hash, PartialEq)]
157pub enum RuleError {
158    /// The requested polynomial degree exceeds the supported maximum.
159    #[error("polynomial degree {degree} exceeds the supported maximum of {maximum}")]
160    DegreeOutOfRange {
161        /// Requested polynomial degree.
162        degree: usize,
163        /// Largest degree available from the constructor or rule set.
164        maximum: usize,
165    },
166    /// The requested point count is outside the supported range.
167    #[error("{family} point count {point_count} is outside supported range {minimum}..={maximum}")]
168    PointCountOutOfRange {
169        /// Quadrature family for which a rule was requested.
170        family: GaussFamily,
171        /// Requested number of points.
172        point_count: usize,
173        /// Smallest supported number of points.
174        minimum: usize,
175        /// Largest supported number of points.
176        maximum: usize,
177    },
178    /// The eigendecomposition backend could not generate a rule.
179    #[error("failed to generate a {family} rule with {point_count} points: {message}")]
180    GenerationFailed {
181        /// Quadrature family being generated.
182        family: GaussFamily,
183        /// Number of points in the requested rule.
184        point_count: usize,
185        /// Deterministic backend diagnostic retained by the cache.
186        message: String,
187    },
188}

191//{{{ struct: GaussRuleSet
192/// A collection of Gauss quadrature rules through a maximum requested exactness degree.
193///
194/// See the [Gaussian-rules guide](https://topohedrallabs.github.io/topohedral-integrate/latest/user-guide/gaussian-rules/).
195#[derive(Clone, Debug, PartialEq)]
196pub struct GaussRuleSet {
197    family: GaussFamily,
198    maximum_degree: PolynomialDegree,
199    rules: Vec<GaussRule>,
200}
201//}}}
202//{{{ impl: GaussRuleSet
203impl GaussRuleSet {
204    /// Builds every rule for `family` through the requested polynomial exactness `degree`.
205    ///
206    /// # Errors
207    ///
208    /// Returns [`RuleError::DegreeOutOfRange`] when `degree` exceeds 100, or
209    /// [`RuleError::GenerationFailed`] if the eigendecomposition backend fails.
210    pub fn through_degree(
211        family: GaussFamily,
212        degree: PolynomialDegree,
213    ) -> Result<Self, RuleError> {
214        validate_degree(degree.value(), MAX_DEGREE)?;
215
216        let minimum = family.minimum_point_count();
217        let maximum = family.point_count_for_degree(degree.value());
218        let mut rules = Vec::with_capacity(maximum - minimum + 1);
219        for point_count in minimum..=maximum {
220            rules.push(build_gauss_rule(family, point_count)?);
221        }
222
223        Ok(Self {
224            family,
225            maximum_degree: degree,
226            rules,
227        })
228    }
229
230    /// Returns the smallest stored rule whose exactness is at least `degree`.
231    ///
232    /// # Errors
233    ///
234    /// Returns [`RuleError::DegreeOutOfRange`] when `degree` exceeds the maximum requested while
235    /// constructing this set.
236    pub fn rule_for_degree(
237        &self,
238        degree: PolynomialDegree,
239    ) -> Result<&GaussRule, RuleError> {
240        self.rule_for_degree_value(degree.value())
241    }
242
243    /// Returns the stored rule containing exactly `point_count` points.
244    ///
245    /// # Errors
246    ///
247    /// Returns [`RuleError::PointCountOutOfRange`] when `point_count` is not represented by this
248    /// set.
249    pub fn rule_by_point_count(
250        &self,
251        point_count: PointCount,
252    ) -> Result<&GaussRule, RuleError> {
253        self.rule_by_point_count_value(point_count.value())
254    }
255
256    fn rule_for_degree_value(
257        &self,
258        degree: usize,
259    ) -> Result<&GaussRule, RuleError> {
260        validate_degree(degree, self.maximum_degree.value())?;
261        self.rule_by_point_count_value(self.family.point_count_for_degree(degree))
262    }
263
264    fn rule_by_point_count_value(
265        &self,
266        point_count: usize,
267    ) -> Result<&GaussRule, RuleError> {
268        let minimum = self.family.minimum_point_count();
269        let maximum = self.maximum_point_count();
270        if !(minimum..=maximum.value()).contains(&point_count) {
271            return Err(RuleError::PointCountOutOfRange {
272                family: self.family,
273                point_count,
274                minimum,
275                maximum: maximum.value(),
276            });
277        }
278        Ok(&self.rules[point_count - minimum])
279    }
280
281    /// Returns the quadrature family shared by every rule in this set.
282    pub fn family(&self) -> GaussFamily {
283        self.family
284    }
285
286    /// Returns the largest requested polynomial degree represented by this set.
287    pub fn maximum_degree(&self) -> PolynomialDegree {
288        self.maximum_degree
289    }
290
291    /// Returns the smallest point count represented by this set.
292    pub fn minimum_point_count(&self) -> PointCount {
293        PointCount::new_unchecked(self.family.minimum_point_count())
294    }
295
296    /// Returns the largest point count represented by this set.
297    pub fn maximum_point_count(&self) -> PointCount {
298        PointCount::new_unchecked(
299            self.family
300                .point_count_for_degree(self.maximum_degree.value()),
301        )
302    }
303
304    /// Returns the number of rules in this set.
305    pub fn len(&self) -> usize {
306        self.rules.len()
307    }
308
309    /// Returns `true` when this set contains no rules.
310    ///
311    /// Generated rule sets are never empty; this method accompanies [`Self::len`] for collection
312    /// API consistency.
313    pub fn is_empty(&self) -> bool {
314        self.rules.is_empty()
315    }
316
317    /// Iterates over the stored rules in ascending point-count order.
318    pub fn iter(&self) -> std::slice::Iter<'_, GaussRule> {
319        self.rules.iter()
320    }
321}

326/// A specific Gauss quadrature rule, represented by points and associated weights.
327///
328/// See the [Gaussian-rules guide](https://topohedrallabs.github.io/topohedral-integrate/latest/user-guide/gaussian-rules/).
329#[derive(Clone, Debug, PartialEq)]
330pub struct GaussRule {
331    family: GaussFamily,
332    exactness: PolynomialDegree,
333    points: Vec<f64>,
334    weights: Vec<f64>,
335}

337//{{{ impl: GaussRule
338impl GaussRule {
339    fn from_points_weights(
340        family: GaussFamily,
341        points: Vec<f64>,
342        weights: Vec<f64>,
343    ) -> Result<Self, RuleError> {
344        if points.len() != weights.len() {
345            return Err(RuleError::GenerationFailed {
346                family,
347                point_count: points.len(),
348                message: format!(
349                    "backend produced {} points but {} weights",
350                    points.len(),
351                    weights.len()
352                ),
353            });
354        }
355        let exactness = family
356            .exactness_for_point_count(points.len())
357            .ok_or_else(|| RuleError::GenerationFailed {
358                family,
359                point_count: points.len(),
360                message: "backend produced an invalid point count".to_owned(),
361            })?;
362        Ok(Self {
363            family,
364            exactness: PolynomialDegree::new_unchecked(exactness),
365            points,
366            weights,
367        })
368    }
369
370    /// Constructs the smallest quadrature rule with at least `degree` polynomial exactness.
371    ///
372    /// Legendre rules support one point and Lobatto rules support two points at minimum. An even
373    /// requested degree selects a rule whose actual exactness is one degree higher.
374    ///
375    /// # Errors
376    ///
377    /// Returns [`RuleError::DegreeOutOfRange`] when `degree` exceeds 100, or
378    /// [`RuleError::GenerationFailed`] if the eigendecomposition backend fails.
379    pub fn for_degree(
380        family: GaussFamily,
381        degree: PolynomialDegree,
382    ) -> Result<Self, RuleError> {
383        validate_degree(degree.value(), MAX_DEGREE)?;
384        build_gauss_rule(family, family.point_count_for_degree(degree.value()))
385    }
386
387    /// Constructs a quadrature rule containing exactly `point_count` points.
388    ///
389    /// # Errors
390    ///
391    /// Returns [`RuleError::PointCountOutOfRange`] when the count is outside the supported range,
392    /// or [`RuleError::GenerationFailed`] if the eigendecomposition backend fails.
393    pub fn with_point_count(
394        family: GaussFamily,
395        point_count: PointCount,
396    ) -> Result<Self, RuleError> {
397        validate_point_count(family, point_count.value())?;
398        build_gauss_rule(family, point_count.value())
399    }
400
401    /// Returns the quadrature family used to construct this rule.
402    pub fn family(&self) -> GaussFamily {
403        self.family
404    }
405
406    /// Returns the rule's actual polynomial exactness.
407    pub fn exactness(&self) -> PolynomialDegree {
408        self.exactness
409    }
410
411    /// Returns the number of quadrature points and weights.
412    pub fn point_count(&self) -> PointCount {
413        PointCount::new_unchecked(self.points.len())
414    }
415
416    /// Returns the quadrature points in ascending order.
417    pub fn points(&self) -> &[f64] {
418        &self.points
419    }
420
421    /// Returns the weights corresponding to [`Self::points`].
422    pub fn weights(&self) -> &[f64] {
423        &self.weights
424    }
425}

536//{{{ fun: build_gauss_rule
537/// Builds a quadrature rule with exactly `point_count` points.
538fn build_gauss_rule(
539    family: GaussFamily,
540    point_count: usize,
541) -> Result<GaussRule, RuleError> {
542    validate_point_count(family, point_count)?;
543
544    let (points, weights) = match family {
545        GaussFamily::Legendre => golub_welsch(
546            family,
547            point_count,
548            point_count,
549            family.weight_integral(),
550            legendre_recursion_coeffs,
551        )?,
552        GaussFamily::Lobatto => {
553            let endpoint_weight = 2.0 / ((point_count as f64) * ((point_count - 1) as f64));
554            let mut points = Vec::with_capacity(point_count);
555            let mut weights = Vec::with_capacity(point_count);
556
557            points.push(-1.0);
558            weights.push(endpoint_weight);
559
560            if point_count > 2 {
561                let interior_count = point_count - 2;
562                let (interior_points, _) = golub_welsch(
563                    family,
564                    point_count,
565                    interior_count,
566                    family.weight_integral(),
567                    lobatto_recursion_coeffs,
568                )?;
569
570                for point in interior_points {
571                    let polynomial = legendre(point_count - 1, point);
572                    points.push(point);
573                    weights.push(endpoint_weight / polynomial.powi(2));
574                }
575            }
576
577            points.push(1.0);
578            weights.push(endpoint_weight);
579            (points, weights)
580        }
581    };
582
583    GaussRule::from_points_weights(family, points, weights)
584}

586//{{{ fun: golub_welsch
587/// Computes the Golub-Welsch algorithm to generate Gauss quadrature points and weights for
588/// numerical integration.
589///
590/// # Arguments
591///
592/// * `family` - type of quadrature rule
593/// * `rule_point_count` - number of points in the final quadrature rule
594/// * `matrix_size` - number of points generated by this eigendecomposition
595/// * `weight_integral` - integral of the recurrence's weight function
596/// * `recurrence_fcn` - recurrence function which provided the recurrence coefficients for the
597///                      orthogonal polynomial family
598///
599/// # Returns
600/// A tuple of two vectors, the first being the quadrature points and the second being the quadrature
601/// weights.
602///
603/// # Theory
604///
605/// Starting from the 3-term orthogonal polynomial recurrence relation:
606/// \\[
607///     p_{i+1}(x) = (a_{i}x + b_{i})p_{i}(x) - c_{i}p_{i-1}
608/// \\]
609/// And rearranging to give:
610/// \\[
611///     xp_{i}(x) =
612///         -\frac{c_{i}}{a_{i}}p_{i-1}(x) + \frac{b_{i}}{a_{i}}p_{i}(x) + \frac{1}{a_{i}}p_{i+1}(x)
613/// \\]
614/// Which may in turn be represented with the following system of equations
615/// \\[
616///     x
617///     \begin{bmatrix}
618///         p_{0}(x) \\\\ p_{1}(x) \\\\ \vdots \\\\ p_{n-2} \\\\ p_{n-1}(x)
619///     \end{bmatrix}
620///     =
621///     \begin{bmatrix}
622///         -b_{1}/a_{1} & 1 / a_{1} & 0 & ... & 0 \\\\
623///         c_{2}/a_{2}  & -b_{2} / a_{2} & 1 / a_{2}   & ... & 0 \\\\
624///         \vdots    &    \ddots      &    \ddots & \ddots  & \vdots \\\\
625///         0  & ... & c_{n-1}/a_{n-1}  & -b_{n-1} / a_{n-1} & 1 / a_{n-1} \\\\
626///         0    &   ... &   0     & c_{n}/a_{n} & -b_{n} \ a_{n}
627///     \end{bmatrix}
628///     \begin{bmatrix}
629///         p_{0}(x) \\\\ p_{1}(x) \\\\ \vdots \\\\ p_{n-2} \\\\ p_{n-1}(x)
630///     \end{bmatrix}
631///     +
632///     \begin{bmatrix}
633///         0 \\\\ 0 \\\\ \vdots \\\\ 0 \\\\ p_{n}(x) / a_{n}
634///     \end{bmatrix}
635/// \\]
636/// Which may be written as:
637/// \\[
638///     x\mathbf{p}(x) = \mathbf{T}\mathbf{p}(x) + \frac{p_{n}(x)}{a_{n}}\mathbf{e_{n}}
639/// \\]
640/// For each of the roots of orthogonal polynomials, which are the quadrature
641/// points of an n-point rule, the following eigenvalue problem is
642/// created:
643/// \\[
644///   t_{j}\mathbf{p}(t_{j}) = \mathbf{T}\mathbf{p}(t_{j})
645/// \\]
646/// The above system is then converted to the following symmetric
647/// system by diagonal similarity transform which preserves the eigenvalues
648/// \\[
649///   t_{j}\mathbf{q}(t_{j}) = \mathbf{J} \mathbf{q}(t_{j})
650/// \\]
651///
652/// Where:
653///
654/// \\[
655///     \mathbf{J} =
656///         \begin{bmatrix}
657///             \alpha_{1} & \beta_{1} & 0 & ... & 0 \\\\
658///              \beta_{1} & \alpha_{2} & \beta_{2} & ... & 0 \\\\
659///              \vdots & \ddots & \ddots  & \ddots & \vdots \\\\
660///              0 & ... & \beta_{n-2} & \alpha_{n-1} & \beta_{n-1} \\\\
661///              0 & 0 & ... & \beta_{n-1} & \alpha_{n}
662///         \end{bmatrix}
663/// \\]
664///
665/// where:
666///
667/// \\[
668///   \alpha_{i} = -\frac{b_{i}}{a_{i}}, \quad
669///   \beta_{i} = \left( \frac{c_{i+1}}{a_{i}a_{i+1}}\right)^{1/2}
670/// \\]
671#[allow(clippy::doc_overindented_list_items)]
672fn golub_welsch<F: Fn(usize) -> (f64, f64, f64)>(
673    family: GaussFamily,
674    rule_point_count: usize,
675    matrix_size: usize,
676    weight_integral: f64,
677    recurrence_fcn: F,
678) -> Result<(Vec<f64>, Vec<f64>), RuleError> {
679    if matrix_size == 0 {
680        return Err(RuleError::GenerationFailed {
681            family,
682            point_count: rule_point_count,
683            message: "eigendecomposition matrix must be nonempty".to_owned(),
684        });
685    }
686
687    let mut tmat = DMatrix::<f64>::zeros(matrix_size, matrix_size);
688
689    for i in 0..matrix_size {
690        let (ai, bi, _) = recurrence_fcn(i);
691        tmat[(i, i)] = -(bi / ai);
692
693        if i + 1 < matrix_size {
694            let (aj, _, cj) = recurrence_fcn(i + 1);
695            let beta = (cj / (ai * aj)).sqrt();
696            tmat[(i, i + 1)] = beta;
697            tmat[(i + 1, i)] = beta;
698        }
699    }
700
701    //{{{ com: eigendecompose
702    let eigen_decomp = tmat.symeig().map_err(|error| RuleError::GenerationFailed {
703        family,
704        point_count: rule_point_count,
705        message: error.to_string(),
706    })?;
707    //}}}
708    //{{{ com: compute quadrature points and weights from eigenvalues and eigenvectors
709    let qpoints: &Vec<f64> = &eigen_decomp.eigvals;
710    let qweights: Vec<f64> = eigen_decomp
711        .eigvecs
712        .row(0)
713        .iter()
714        .map(|x| x.powi(2) * weight_integral)
715        .collect();
716    let mut combined: Vec<(f64, f64)> = qpoints
717        .iter()
718        .cloned()
719        .zip(qweights.iter().cloned())
720        .collect();
721    combined.sort_by(|a, b| a.0.total_cmp(&b.0));
722    let (qpoints_final, qweights_final): (Vec<f64>, Vec<f64>) = combined.iter().cloned().unzip();
723    //}}}
724    //{{{ ret
725    Ok((qpoints_final, qweights_final))
726    //}}}
727}

730//{{{ fun: legendre_recursion_coeffs
731/// Computes the recurrence coefficients for the nth Legendre polynomial using the recurrence relation.
732///
733/// Returns a tuple containing the coefficients (a, b, c) for the i'th polynomial.
734fn legendre_recursion_coeffs(i: usize) -> (f64, f64, f64) {
735    let i_f64 = i as f64;
736    let ai = (2.0 * i_f64 + 1.0) / (i_f64 + 1.0);
737    let bi = 0.0;
738    let ci = i_f64 / (i_f64 + 1.0);
739    (ai, bi, ci)
740}
741//}}}
742//{{{ fun: lobatto_recursion_coeffs
743/// Computes the recurrence coefficients `ai`, `bi`, and `ci` for
744/// the Lobatto quadrature rule at index `i`.
745///
746/// Returns a tuple containing the coefficients (a, b, c) for the i'th polynomial.
747fn lobatto_recursion_coeffs(i: usize) -> (f64, f64, f64) {
748    let i_f64 = i as f64;
749    let ai = ((2.0 * i_f64 + 3.0) * (i_f64 + 2.0)) / ((i_f64 + 1.0) * (i_f64 + 3.0));
750    let bi = 0.0;
751    let ci = ((i_f64 + 1.0) * (i_f64 + 2.0)) / ((i_f64 + 3.0) * (i_f64 + 1.0));
752    (ai, bi, ci)
753}
754//}}}
755//{{{ fun: legendre
756fn legendre(
757    n: usize,
758    x: f64,
759) -> f64 {
760    let (mut leg_n, mut leg_1, mut leg_2) = (1.0f64, 1.0f64, 0.0f64);
761    for i in 0..n {
762        let ii = i as f64;
763        let ai = (2.0 * ii + 1.0) / (ii + 1.0);
764        let ci = ii / (ii + 1.0);
765        leg_n = ai * x * leg_1 - ci * leg_2;
766        leg_2 = leg_1;
767        leg_1 = leg_n;
768    }
769    leg_n
770}
