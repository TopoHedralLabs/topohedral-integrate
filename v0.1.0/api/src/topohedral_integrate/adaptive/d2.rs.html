22//{{{ collection: Builder
23/// Consuming builder for two-dimensional adaptive quadrature.
24///
25/// See the [adaptive-quadrature guide](https://topohedrallabs.github.io/topohedral-integrate/latest/user-guide/adaptive-quadrature/).
26#[derive(Clone, Debug, PartialEq)]
27pub struct Builder {
28    domain: Rectangle,
29    low_rule: TensorRule,
30    high_rule: TensorRule,
31    tolerance: Tolerance,
32    max_depth: AxisDepths,
33    u_subdivisions: Vec<f64>,
34    v_subdivisions: Vec<f64>,
35}
36
37impl Builder {
38    /// Sets the maximum refinement depth independently on each axis.
39    pub const fn max_depth(
40        mut self,
41        max_depth: AxisDepths,
42    ) -> Self {
43        self.max_depth = max_depth;
44        self
45    }
46
47    /// Replaces the initial interior subdivision coordinates on both axes.
48    ///
49    /// # Errors
50    ///
51    /// Returns all detected subdivision validation issues.
52    pub fn subdivisions<U, V>(
53        mut self,
54        u_points: U,
55        v_points: V,
56    ) -> Result<Self, ConfigError>
57    where
58        U: IntoIterator<Item = f64>,
59        V: IntoIterator<Item = f64>,
60    {
61        let u = validate_subdivisions(self.domain.u(), u_points);
62        let v = validate_subdivisions(self.domain.v(), v_points);
63        match (u, v) {
64            (Ok(u), Ok(v)) => {
65                self.u_subdivisions = u;
66                self.v_subdivisions = v;
67                Ok(self)
68            }
69            (Err(u), Err(v)) => {
70                let mut issues = u.issues().to_vec();
71                issues.extend_from_slice(v.issues());
72                Err(ConfigError::from_issues(issues))
73            }
74            (Err(error), _) | (_, Err(error)) => Err(error),
75        }
76    }
77
78    /// Validates both axis rule pairs and constructs a reusable adaptive quadrature.
79    ///
80    /// # Errors
81    ///
82    /// Returns [`ConfigError`] for every axis whose high rule's actual exactness does not exceed
83    /// its low rule's.
84    pub fn build(self) -> Result<AdaptiveQuadrature, ConfigError> {
85        let mut issues = Vec::new();
86        validate_rule_pair(
87            self.low_rule.u(),
88            self.high_rule.u(),
89            RuleAxis::U,
90            &mut issues,
91        );
92        validate_rule_pair(
93            self.low_rule.v(),
94            self.high_rule.v(),
95            RuleAxis::V,
96            &mut issues,
97        );
98        if !issues.is_empty() {
99            return Err(ConfigError::from_issues(issues));
100        }
101
102        Ok(AdaptiveQuadrature {
103            domain: self.domain,
104            low_rule: FixedQuadrature::builder(self.domain, self.low_rule).build(),
105            high_rule: FixedQuadrature::builder(self.domain, self.high_rule).build(),
106            tolerance: self.tolerance,
107            max_depth: self.max_depth,
108            u_subdivisions: self.u_subdivisions,
109            v_subdivisions: self.v_subdivisions,
110        })
111    }
112}

115//{{{ collection: AdaptiveQuadrature
116/// Reusable two-dimensional adaptive tensor-product quadrature configuration.
117///
118/// See the [two-dimensional adaptive example](https://topohedrallabs.github.io/topohedral-integrate/latest/user-guide/adaptive-quadrature/#two-dimensional-integration).
119#[derive(Clone, Debug, PartialEq)]
120pub struct AdaptiveQuadrature {
121    domain: Rectangle,
122    low_rule: FixedQuadrature,
123    high_rule: FixedQuadrature,
124    tolerance: Tolerance,
125    max_depth: AxisDepths,
126    u_subdivisions: Vec<f64>,
127    v_subdivisions: Vec<f64>,
128}
129
130impl AdaptiveQuadrature {
131    /// Starts a consuming builder with a default maximum depth of 32 on both axes.
132    pub fn builder(
133        domain: Rectangle,
134        low_rule: TensorRule,
135        high_rule: TensorRule,
136        tolerance: Tolerance,
137    ) -> Builder {
138        Builder {
139            domain,
140            low_rule,
141            high_rule,
142            tolerance,
143            max_depth: AxisDepths::default(),
144            u_subdivisions: Vec::new(),
145            v_subdivisions: Vec::new(),
146        }
147    }
148
149    /// Returns the complete integration domain.
150    pub const fn domain(&self) -> Rectangle {
151        self.domain
152    }
153
154    /// Returns the low-order tensor-product rule.
155    pub const fn low_rule(&self) -> &TensorRule {
156        self.low_rule.rule()
157    }
158
159    /// Returns the high-order tensor-product rule.
160    pub const fn high_rule(&self) -> &TensorRule {
161        self.high_rule.rule()
162    }
163
164    /// Returns the global convergence tolerance.
165    pub const fn tolerance(&self) -> Tolerance {
166        self.tolerance
167    }
168
169    /// Returns the maximum per-axis refinement depths.
170    pub const fn max_depth(&self) -> AxisDepths {
171        self.max_depth
172    }
173
174    /// Returns the validated initial `u`-axis subdivision coordinates.
175    pub fn u_subdivision_points(&self) -> &[f64] {
176        &self.u_subdivisions
177    }
178
179    /// Returns the validated initial `v`-axis subdivision coordinates.
180    pub fn v_subdivision_points(&self) -> &[f64] {
181        &self.v_subdivisions
182    }
183
184    /// Adaptively integrates `f` using a global error estimate.
185    ///
186    /// Both axes are split while both remain below their limits. Once one axis reaches its
187    /// maximum depth, only the other axis is split.
188    ///
189    /// # Errors
190    ///
191    /// Returns [`IntegrationError::NonFiniteIntegrand`] if `f` returns NaN or infinity,
192    /// [`IntegrationError::MaxDepthReached`] if the tolerance cannot be met within the configured
193    /// per-axis depths, or [`IntegrationError::NonProgressingRectangle`] if an active-axis
194    /// floating-point midpoint is not strictly interior.
195    ///
196    /// # Panics
197    ///
198    /// Panics raised by `f` are not caught and propagate to the caller.
199    ///
200    /// # Examples
201    ///
202    /// ```
203    /// use topohedral_integrate::{
204    ///     AdaptiveQuadrature2d, GaussFamily, GaussRule, PolynomialDegree, Rectangle,
205    ///     TensorRule2d, Tolerance,
206    /// };
207    ///
208    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
209    /// let low = GaussRule::for_degree(GaussFamily::Legendre, PolynomialDegree::new(3)?)?;
210    /// let high = GaussRule::for_degree(GaussFamily::Legendre, PolynomialDegree::new(7)?)?;
211    /// let quadrature = AdaptiveQuadrature2d::builder(
212    ///     Rectangle::from_bounds(0.0, 1.0, 0.0, 1.0)?,
213    ///     TensorRule2d::new(low.clone(), low),
214    ///     TensorRule2d::new(high.clone(), high),
215    ///     Tolerance::absolute(1e-10)?,
216    /// )
217    /// .build()?;
218    /// let result = quadrature.integrate(|u, v| u * u + v * v)?;
219    /// assert!((result.integral() - 2.0 / 3.0).abs() < 1e-12);
220    /// # Ok(())
221    /// # }
222    /// ```
223    pub fn integrate<F>(
224        &self,
225        mut f: F,
226    ) -> Result<AdaptiveResult, IntegrationError>
227    where
228        F: FnMut(f64, f64) -> f64,
229    {
230        let evaluations_per_region = self.low_rule.point_count() + self.high_rule.point_count();
231        let u_bounds = axis_bounds(self.domain.u(), &self.u_subdivisions);
232        let v_bounds = axis_bounds(self.domain.v(), &self.v_subdivisions);
233        let mut regions = Vec::with_capacity((u_bounds.len() - 1) * (v_bounds.len() - 1));
234        let mut evaluation_count = 0;
235
236        for u in u_bounds.windows(2) {
237            for v in v_bounds.windows(2) {
238                let domain = Rectangle::new(
239                    Interval::new_unchecked(u[0], u[1]),
240                    Interval::new_unchecked(v[0], v[1]),
241                );
242                regions.push(evaluate_region(
243                    domain,
244                    0,
245                    0,
246                    &self.low_rule,
247                    &self.high_rule,
248                    &mut f,
249                )?);
250                evaluation_count += evaluations_per_region;
251            }
252        }
253
254        let mut global_integral: f64 = regions.iter().map(|region| region.high_estimate).sum();
255        let mut global_error: f64 = regions.iter().map(|region| region.error_estimate).sum();
256
257        loop {
258            let result = AdaptiveResult::new(
259                global_integral,
260                global_error,
261                regions.len(),
262                evaluation_count,
263            );
264            if converged(&result, self.tolerance) {
265                return Ok(result);
266            }
267
268            let Some(index) = regions
269                .iter()
270                .enumerate()
271                .filter(|(_, region)| self.can_split(region))
272                .max_by(|(_, left), (_, right)| {
273                    left.error_estimate.total_cmp(&right.error_estimate)
274                })
275                .map(|(index, _)| index)
276            else {
277                return Err(IntegrationError::MaxDepthReached {
278                    partial_result: result,
279                });
280            };
281
282            let parent = regions[index];
283            let split_u = parent.u_depth < self.max_depth.u().value();
284            let split_v = parent.v_depth < self.max_depth.v().value();
285            let u_midpoint = split_u.then(|| midpoint(parent.domain.u()));
286            let v_midpoint = split_v.then(|| midpoint(parent.domain.v()));
287            if u_midpoint.is_some_and(|midpoint| {
288                midpoint <= parent.domain.u().lower() || midpoint >= parent.domain.u().upper()
289            }) || v_midpoint.is_some_and(|midpoint| {
290                midpoint <= parent.domain.v().lower() || midpoint >= parent.domain.v().upper()
291            }) {
292                return Err(IntegrationError::NonProgressingRectangle {
293                    rectangle: parent.domain,
294                    partial_result: result,
295                });
296            }
297
298            let child_domains = child_domains(parent.domain, u_midpoint, v_midpoint);
299            let u_depth = parent.u_depth + usize::from(split_u);
300            let v_depth = parent.v_depth + usize::from(split_v);
301            let child_count = child_domains.len();
302            let mut children = Vec::with_capacity(child_count);
303            for domain in child_domains {
304                children.push(evaluate_region(
305                    domain,
306                    u_depth,
307                    v_depth,
308                    &self.low_rule,
309                    &self.high_rule,
310                    &mut f,
311                )?);
312            }
313
314            let child_integral: f64 = children.iter().map(|child| child.high_estimate).sum();
315            let child_error: f64 = children.iter().map(|child| child.error_estimate).sum();
316            evaluation_count += child_count * evaluations_per_region;
317            global_integral += child_integral - parent.high_estimate;
318            global_error = (global_error - parent.error_estimate).max(0.0) + child_error;
319            regions.swap_remove(index);
320            regions.extend(children);
321        }
322    }
323
324    fn can_split(
325        &self,
326        region: &Region,
327    ) -> bool {
328        region.u_depth < self.max_depth.u().value() || region.v_depth < self.max_depth.v().value()
329    }
330}

333//{{{ fun: validate_rule_pair
334fn validate_rule_pair(
335    low: &crate::GaussRule,
336    high: &crate::GaussRule,
337    axis: RuleAxis,
338    issues: &mut Vec<ConfigIssue>,
339) {
340    let low = low.exactness().value();
341    let high = high.exactness().value();
342    if high <= low {
343        issues.push(ConfigIssue::NonIncreasingRuleExactness { axis, low, high });
344    }
345}

348//{{{ fun: evaluate_region
349fn evaluate_region<F>(
350    domain: Rectangle,
351    u_depth: usize,
352    v_depth: usize,
353    low_rule: &FixedQuadrature,
354    high_rule: &FixedQuadrature,
355    f: &mut F,
356) -> Result<Region, IntegrationError>
357where
358    F: FnMut(f64, f64) -> f64,
359{
360    let low_estimate = low_rule.integrate_over(domain, &mut *f)?;
361    let high_estimate = high_rule.integrate_over(domain, &mut *f)?;
362    Ok(Region {
363        domain,
364        u_depth,
365        v_depth,
366        high_estimate,
367        error_estimate: (high_estimate - low_estimate).abs(),
368    })
369}

372//{{{ fun: axis_bounds
373fn axis_bounds(
374    interval: Interval,
375    subdivisions: &[f64],
376) -> Vec<f64> {
377    let mut bounds = Vec::with_capacity(subdivisions.len() + 2);
378    bounds.push(interval.lower());
379    bounds.extend_from_slice(subdivisions);
380    bounds.push(interval.upper());
381    bounds
382}
383//}}}

385//{{{ fun: child_domains
386fn child_domains(
387    domain: Rectangle,
388    u_midpoint: Option<f64>,
389    v_midpoint: Option<f64>,
390) -> Vec<Rectangle> {
391    let u_intervals = split_axis(domain.u(), u_midpoint);
392    let v_intervals = split_axis(domain.v(), v_midpoint);
393    let mut children = Vec::with_capacity(u_intervals.len() * v_intervals.len());
394    for u in &u_intervals {
395        for v in &v_intervals {
396            children.push(Rectangle::new(*u, *v));
397        }
398    }
399    children
400}
401//}}}

403//{{{ fun: split_axis
404fn split_axis(
405    interval: Interval,
406    midpoint: Option<f64>,
407) -> Vec<Interval> {
408    match midpoint {
409        Some(midpoint) => vec![
410            Interval::new_unchecked(interval.lower(), midpoint),
411            Interval::new_unchecked(midpoint, interval.upper()),
412        ],
413        None => vec![interval],
414    }
415}

418//{{{ fun: converged
419fn converged(
420    result: &AdaptiveResult,
421    tolerance: Tolerance,
422) -> bool {
423    result.error_estimate()
424        <= tolerance.absolute_value() + tolerance.relative_value() * result.integral().abs()
425}
426//}}}
427
428//{{{ fun: midpoint
429fn midpoint(interval: Interval) -> f64 {
430    let direct = interval.lower() + 0.5 * (interval.upper() - interval.lower());
431    if direct.is_finite() {
432        direct
433    } else {
434        0.5 * interval.lower() + 0.5 * interval.upper()
435    }
436}

439//{{{ fun: adaptive_quad
440/// Builds a two-dimensional adaptive quadrature and integrates `f` once.
441/// See the [one-shot example](https://topohedrallabs.github.io/topohedral-integrate/latest/user-guide/adaptive-quadrature/#one-shot-helpers).
442///
443/// Use [`AdaptiveQuadrature::builder`] to configure the consuming `builder`. Construct an
444/// [`AdaptiveQuadrature`] directly when the same configuration will be reused.
445///
446/// # Errors
447///
448/// Returns [`OptionsError`] if either axis rule pair is invalid or adaptive integration fails.
449///
450/// # Panics
451///
452/// Panics raised by `f` are not caught and propagate to the caller.
453pub fn adaptive_quad<F>(
454    f: F,
455    builder: Builder,
456) -> Result<AdaptiveResult, OptionsError>
457where
458    F: FnMut(f64, f64) -> f64,
459{
460    Ok(builder.build()?.integrate(f)?)
461}
