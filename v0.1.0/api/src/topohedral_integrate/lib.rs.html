1//! Numerical quadrature rules for one- and two-dimensional real-valued functions.
2//!
3//! The crate provides fixed Gauss quadrature rules and adaptive algorithms built from pairs of
4//! fixed rules. The dimensionality is encoded in the names of the public types and functions.
5//!
6//! # Fixed quadrature
7//!
8//! ```
9//! use topohedral_integrate::{
10//!     FixedQuadrature1d, GaussFamily, GaussRule, Interval, PolynomialDegree,
11//! };
12//!
13//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
14//! let domain = Interval::new(-1.0, 1.0)?;
15//! let degree = PolynomialDegree::new(5)?;
16//! let rule = GaussRule::for_degree(GaussFamily::Legendre, degree)?;
17//! let quadrature = FixedQuadrature1d::builder(domain, rule).build();
18//! let integral = quadrature.integrate(|x| x * x)?;
19//! assert!((integral - 2.0 / 3.0).abs() < 1e-12);
20//! # Ok(())
21//! # }
22//! ```
23//!
24//! # Adaptive quadrature
25//!
26//! ```
27//! use topohedral_integrate::{
28//!     AdaptiveQuadrature1d, GaussFamily, GaussRule, Interval, PolynomialDegree, Tolerance,
29//! };
30//!
31//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
32//! let domain = Interval::new(0.0, 1.0)?;
33//! let low = GaussRule::for_degree(GaussFamily::Legendre, PolynomialDegree::new(3)?)?;
34//! let high = GaussRule::for_degree(GaussFamily::Legendre, PolynomialDegree::new(9)?)?;
35//! let tolerance = Tolerance::new(1e-10, 1e-10)?;
36//! let quadrature = AdaptiveQuadrature1d::builder(domain, low, high, tolerance).build()?;
37//! let result = quadrature.integrate(|x| x.exp())?;
38//! assert!((result.integral() - (std::f64::consts::E - 1.0)).abs() < 1e-9);
39//! # Ok(())
40//! # }
41//! ```
42//!
43//! The optional `serde` feature serializes validated values, generated rules, nodes, errors, and
44//! adaptive results. Deserialization re-applies constructor validation. The optional `trace`
45//! feature enables instrumentation without enabling terminal colours.
46//!
